    \"\"\"Calculates the area of a right-angled triangle given base and height.\"\"\"

def is_palindrome(number: int) -> bool:
    \"\"\"Checks if a non-negative integer is a palindrome.\"\"\"
    if number < 0:
        return False
    
    original = number
    reversed_num = 0
    while number > 0:
        remainder = number % 10
        reversed_num = (reversed_num * 10) + remainder
        number //= 10
        
    return original == reversed_num

def main():
    try:
        num = int(input("Enter a positive integer: "))
        if is_palindrome(num):
            print(f"[+] The number {num} is a Palindrome.")
        else:
            print(f"[-] The number {num} is NOT a Palindrome.")
    except ValueError:
        print("[!] Input error: Please enter a valid integer.")

if __name__ == "__main__":
    main()
\end{lstlisting}

from typing import List

def generate_fibonacci(n_terms: int) -> List[int]:
    \"\"\"Generates a list containing the first n terms of the Fibonacci sequence.\"\"\"
    if n_terms <= 0:
        return []
    elif n_terms == 1:
        return [0]
    
    series = [0, 1]
    while len(series) < n_terms:
        next_term = series[-1] + series[-2]
        series.append(next_term)
    return series

def main():
    try:
        n = int(input("Enter number of terms to generate: "))
        if n <= 0:
            print("[!] Please enter a positive integer greater than zero.")
            return
        
        result = generate_fibonacci(n)
        print(f"\n[+] Fibonacci Series ({n} terms):")
        print(" -> ".join(map(str, result)))
    except ValueError:
        print("[!] Input error: Please enter a valid integer.")

if __name__ == "__main__":
    main()
\end{lstlisting}

from typing import Tuple

def count_vowels_and_consonants(text: str) -> Tuple[int, int]:
    \"\"\"Counts vowels and consonants in a given string.\"\"\"
    VOWELS = set("aeiouAEIOU")
    vowels_count = 0
    consonants_count = 0
    
    for char in text:
        if char.isalpha():
            if char in VOWELS:
                vowels_count += 1
            else:
                consonants_count += 1
                
    return vowels_count, consonants_count

def main():
    user_input = input("Enter a string for character analysis: ")
    v_count, c_count = count_vowels_and_consonants(user_input)
    
    print(f"\n[+] String Analyzed: \"{user_input}\"")
    print(f"[+] Total Vowels     : {v_count}")
    print(f"[+] Total Consonants : {c_count}")

if __name__ == "__main__":
    main()

def bubble_sort(arr): 
    n = len(arr) 
    for i in range(n): 
        for j in range(0, n - i - 1): 
            if arr[j] > arr[j + 1]:

\begin{lstlisting}[style=pythonstyle]
from typing import List

def bubble_sort_optimized(arr: List[int]) -> List[int]:
    \"\"\"Sorts a list of numbers in non-decreasing order using optimized Bubble Sort.\"\"\"
    data = arr.copy()
    n = len(data)
    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            if data[j] > data[j + 1]:
                data[j], data[j + 1] = data[j + 1], data[j]
                swapped = True
        if not swapped:
            break  # Early termination if list is already sorted
    return data

def main():
    sample_list = [64, 34, 25, 12, 22, 11, 90]
    print(f"[+] Original List : {sample_list}")
    sorted_res = bubble_sort_optimized(sample_list)
    print(f"[+] Sorted List   : {sorted_res}")

if __name__ == "__main__":
    main()

from collections import Counter
from typing import List, Any

def get_descending_frequencies(elements: List[Any]) -> List[int]:
    \"\"\"Returns element occurrence frequencies sorted in non-increasing (descending) order.\"\"\"
    if not elements:
        return []
    
    # Use standard library Counter for O(N) frequency tally
    freq_map = Counter(elements)
    # Extract frequencies and sort descending
    return sorted(freq_map.values(), reverse=True)

import matplotlib.pyplot as plt
import matplotlib.patches as patches
import numpy as np

def draw_styled_geometric_shapes():
    \"\"\"Renders styled geometric shapes with legend using Matplotlib patches.\"\"\"
    fig, ax = plt.subplots(figsize=(7, 7))

    # Add shapes
    rect = patches.Rectangle((0.15, 0.45), 0.35, 0.3, color='#1F77B4', alpha=0.6, label='Rectangle')
    circ = patches.Circle((0.75, 0.25), 0.15, color='#2CA02C', alpha=0.6, label='Circle')
    tri_pts = np.array([[0.55, 0.75], [0.85, 0.75], [0.70, 0.95]])
    poly = patches.Polygon(tri_pts, closed=True, color='#D62728', alpha=0.6, label='Triangle')

    ax.add_patch(rect)
    ax.add_patch(circ)
    ax.add_patch(poly)

    # Plot styling
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect('equal')
    ax.grid(True, linestyle='--', alpha=0.5)
    ax.set_xlabel('X Coordinate', fontsize=11, fontweight='bold')
    ax.set_ylabel('Y Coordinate', fontsize=11, fontweight='bold')
    ax.set_title('Geometric Shapes Visualization', fontsize=14, fontweight='bold', pad=15)
    ax.legend(loc='upper left')

    plt.tight_layout()
    plt.show()

if __name__ == "__main__":
    draw_styled_geometric_shapes()
\end{lstlisting}
